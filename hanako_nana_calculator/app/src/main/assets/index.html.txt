}
function calculate(s){

 const peek=()=>t[p],take=()=>t[p++];
 function primary(){

 }
 function postfix(){let v=primary();while(peek()==='!'||peek()==='%'){let op=take();if(op==='!'){if(v<0||v>170||!Number.isInteger(v))throw Error('팩토리얼 범위 오류');let r=1;for(let i=2;i<=v;i++)r*=i;v=r}else v/=100}return v}
 function power(){let v=postfix();if(peek()==='^'){take();v=Math.pow(v,power())}return v}
 function mul(){let v=power();while(peek()==='*'||peek()==='/'){let op=take(),r=power();v=op==='*'?v*r:v/r}return v}
 function add(){let v=mul();while(peek()==='+'||peek()==='-'){let op=take(),r=mul();v=op==='+'?v+r:v-r}return v}

}
